package com.example.contactmasksafe;

import android.accessibilityservice.AccessibilityService;
import android.accessibilityservice.AccessibilityServiceInfo;
import android.content.res.Configuration;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.graphics.PixelFormat;
import android.graphics.Rect;
import android.graphics.RectF;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.os.SystemClock;
import android.view.Gravity;
import android.view.View;
import android.view.WindowManager;
import android.view.accessibility.AccessibilityEvent;
import android.view.accessibility.AccessibilityNodeInfo;
import android.view.accessibility.AccessibilityWindowInfo;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Locale;
import java.util.Set;

/**
 * Universal, anti-blink and ScrollSafe saved-number masking engine.
 * The non-touchable overlay remains attached while Privacy Shield is enabled.
 */
public class PrivacyAccessibilityService extends AccessibilityService {
    private static final int MAX_NODES = 4200;
    private static final int FAST_EVENT_NODES = 260;
    private static final int MAX_MASKS = 120;
    private static final long EMPTY_FRAME_HOLD_MS = 1900L;
    private static final long SCROLL_SETTLE_MS = 300L;
    private static final long ACTIVE_RESCAN_MS = 28L;
    private static final long HEALTH_SCAN_MS = 900L;
    private static final long CONTACT_REFRESH_MS = 30000L;
    private static final long TRANSITION_CURTAIN_MS = 500L;
    private static final int AXIS_VERTICAL = 1;
    private static final int AXIS_HORIZONTAL = 2;

    private final Handler handler = new Handler(Looper.getMainLooper());
    private final List<MaskRegion> stableMasks = new ArrayList<>();
    private SavedNumberRepository repository;
    private WindowManager windowManager;
    private MaskLayerView maskLayerView;
    private boolean overlayAttached;
    private boolean scanQueued;
    private boolean transitionCurtain;
    private long lastConfirmedMaskAt;
    private long scrollGuardUntil;
    private long lastContactRefreshAt;
    private int scrollAxis = AXIS_VERTICAL;
    private String foregroundPackage = "";

    private final Runnable scanRunnable = new Runnable() {
        @Override public void run() { scanQueued = false; scanVisibleWindows(); }
    };
    private final Runnable activeMotionScan = new Runnable() {
        @Override public void run() {
            if (!AppPreferences.isPrivacyShieldEnabled(PrivacyAccessibilityService.this)) return;
            long now = SystemClock.uptimeMillis();
            scanVisibleWindows();
            if (now < scrollGuardUntil) handler.postDelayed(this, ACTIVE_RESCAN_MS);
            else { render(); scheduleScan(0L); scheduleScan(90L); }
        }
    };
    private final Runnable healthScan = new Runnable() {
        @Override public void run() {
            if (AppPreferences.isPrivacyShieldEnabled(PrivacyAccessibilityService.this)) {
                if (!stableMasks.isEmpty() || isPhoneOrContactsPackage(foregroundPackage)) scanVisibleWindows();
                handler.postDelayed(this, HEALTH_SCAN_MS);
            }
        }
    };
    private final Runnable clearCurtain = new Runnable() {
        @Override public void run() { transitionCurtain = false; render(); }
    };

    @Override protected void onServiceConnected() {
        super.onServiceConnected();
        repository = new SavedNumberRepository(this);
        repository.refreshNow();
        lastContactRefreshAt = SystemClock.uptimeMillis();
        windowManager = (WindowManager) getSystemService(WINDOW_SERVICE);
        AccessibilityServiceInfo info = getServiceInfo();
        if (info != null) {
            info.flags |= AccessibilityServiceInfo.FLAG_REPORT_VIEW_IDS;
            info.flags |= AccessibilityServiceInfo.FLAG_RETRIEVE_INTERACTIVE_WINDOWS;
            info.flags |= AccessibilityServiceInfo.FLAG_INCLUDE_NOT_IMPORTANT_VIEWS;
            info.notificationTimeout = 0;
            setServiceInfo(info);
        }
        AppPreferences.recordServiceConnected(this);
        if (AppPreferences.isPrivacyShieldEnabled(this)) {
            ensureOverlayAttached();
            scheduleScan(0L);
            handler.removeCallbacks(healthScan);
            handler.postDelayed(healthScan, HEALTH_SCAN_MS);
        }
    }

    @Override public void onAccessibilityEvent(AccessibilityEvent event) {
        if (!AppPreferences.isPrivacyShieldEnabled(this)) { detachOverlay(); return; }
        ensureOverlayAttached();
        refreshContactsIfNeeded();
        String oldPackage = foregroundPackage;
        if (event != null && event.getPackageName() != null) foregroundPackage = event.getPackageName().toString();
        boolean packageChanged = !oldPackage.isEmpty() && !foregroundPackage.isEmpty() && !oldPackage.equals(foregroundPackage);
        int type = event == null ? 0 : event.getEventType();
        boolean windowChanged = type == AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED || type == AccessibilityEvent.TYPE_WINDOWS_CHANGED;
        if ((packageChanged || windowChanged) && isPhoneOrContactsPackage(foregroundPackage) && stableMasks.isEmpty()) beginTransitionCurtain();
        if (type == AccessibilityEvent.TYPE_VIEW_SCROLLED) beginScrollGuard(event);
        promoteMasksFromEvent(event);
        scheduleScan(0L);
        scheduleScan(55L);
        scheduleScan(180L);
    }

    private void beginScrollGuard(AccessibilityEvent event) {
        long now = SystemClock.uptimeMillis();
        scrollGuardUntil = Math.max(scrollGuardUntil, now + SCROLL_SETTLE_MS);
        scrollAxis = AXIS_VERTICAL;
        if (Build.VERSION.SDK_INT >= 28 && event != null) {
            try {
                int dx = Math.abs(event.getScrollDeltaX());
                int dy = Math.abs(event.getScrollDeltaY());
                if (dx > dy) scrollAxis = AXIS_HORIZONTAL;
            } catch (RuntimeException ignored) { }
        }
        if (stableMasks.isEmpty() && isPhoneOrContactsPackage(foregroundPackage)) beginTransitionCurtain(180L);
        render();
        handler.removeCallbacks(activeMotionScan);
        handler.postAtFrontOfQueue(activeMotionScan);
    }

    private boolean isScrolling() { return SystemClock.uptimeMillis() < scrollGuardUntil; }

    private void scheduleScan(long delay) {
        if (delay <= 0L) {
            if (scanQueued) return;
            scanQueued = true;
            handler.postAtFrontOfQueue(scanRunnable);
        } else handler.postDelayed(() -> scanVisibleWindows(), delay);
    }

    private void promoteMasksFromEvent(AccessibilityEvent event) {
        if (event == null || repository == null) return;
        AccessibilityNodeInfo source = event.getSource();
        if (source == null) return;
        List<MaskRegion> found = new ArrayList<>();
        Set<String> keys = new HashSet<>();
        collectMasks(source, found, keys, FAST_EVENT_NODES);
        if (!found.isEmpty()) {
            if (isScrolling()) mergeDuringMotion(found); else replaceMasks(found);
            lastConfirmedMaskAt = SystemClock.uptimeMillis();
            endTransitionCurtain();
            render();
        }
    }

    private void scanVisibleWindows() {
        if (!AppPreferences.isPrivacyShieldEnabled(this) || repository == null || windowManager == null) {
            detachOverlay();
            AppPreferences.recordAccessibilityEvent(this, 0);
            return;
        }
        ensureOverlayAttached();
        refreshContactsIfNeeded();
        List<MaskRegion> found = new ArrayList<>();
        Set<String> keys = new HashSet<>();
        int remaining = MAX_NODES;
        try {
            List<AccessibilityWindowInfo> windows = getWindows();
            if (windows != null && !windows.isEmpty()) {
                for (AccessibilityWindowInfo w : windows) {
                    if (w == null || remaining <= 0 || found.size() >= MAX_MASKS) continue;
                    AccessibilityNodeInfo root = w.getRoot();
                    if (root == null) continue;
                    remaining -= collectMasks(root, found, keys, remaining);
                }
            } else {
                AccessibilityNodeInfo root = getRootInActiveWindow();
                if (root != null) collectMasks(root, found, keys, remaining);
            }
        } catch (RuntimeException ignored) { }
        long now = SystemClock.uptimeMillis();
        if (!found.isEmpty()) {
            if (isScrolling()) mergeDuringMotion(found); else replaceMasks(found);
            lastConfirmedMaskAt = now;
            endTransitionCurtain();
        } else if (!stableMasks.isEmpty() && (isScrolling() || now - lastConfirmedMaskAt <= EMPTY_FRAME_HOLD_MS)) {
            // Retain the last confirmed protection through short OEM/animation empty frames.
        } else if (!isScrolling()) stableMasks.clear();
        render();
        AppPreferences.recordAccessibilityEvent(this, stableMasks.size());
    }

    private int collectMasks(AccessibilityNodeInfo root, List<MaskRegion> out, Set<String> keys, int limit) {
        if (root == null || limit <= 0) return 0;
        ArrayDeque<AccessibilityNodeInfo> q = new ArrayDeque<>();
        q.add(root);
        int visited = 0;
        while (!q.isEmpty() && visited < limit && out.size() < MAX_MASKS) {
            AccessibilityNodeInfo node = q.removeFirst();
            visited++;
            try {
                if (!node.isVisibleToUser()) continue;
                boolean sensitive = PhoneMasker.containsSavedNumber(node.getText(), repository)
                        || PhoneMasker.containsSavedNumber(node.getContentDescription(), repository);
                if (Build.VERSION.SDK_INT >= 26) sensitive = sensitive || PhoneMasker.containsSavedNumber(node.getHintText(), repository);
                if (sensitive) {
                    Rect b = new Rect();
                    node.getBoundsInScreen(b);
                    addRegion(out, keys, b);
                }
                int count = node.getChildCount();
                for (int i = 0; i < count && visited + q.size() < limit && out.size() < MAX_MASKS; i++) {
                    AccessibilityNodeInfo child = node.getChild(i);
                    if (child != null) q.addLast(child);
                }
            } catch (RuntimeException ignored) {
            } finally { try { node.recycle(); } catch (RuntimeException ignored) { } }
        }
        while (!q.isEmpty()) { try { q.removeFirst().recycle(); } catch (RuntimeException ignored) { } }
        return visited;
    }

    private void addRegion(List<MaskRegion> out, Set<String> keys, Rect b) {
        if (b == null || b.width() < 8 || b.height() < 6) return;
        String key = (b.left / 4) + ":" + (b.top / 4) + ":" + (b.right / 4) + ":" + (b.bottom / 4);
        if (keys.add(key)) out.add(new MaskRegion(new Rect(b)));
    }
    private void replaceMasks(List<MaskRegion> found) { stableMasks.clear(); stableMasks.addAll(found); }
    private void mergeDuringMotion(List<MaskRegion> found) {
        for (MaskRegion fresh : found) {
            boolean matched = false;
            for (int i = 0; i < stableMasks.size(); i++) {
                Rect old = stableMasks.get(i).bounds;
                if (Rect.intersects(old, fresh.bounds) || (Math.abs(old.centerX() - fresh.bounds.centerX()) < 28 && Math.abs(old.centerY() - fresh.bounds.centerY()) < 90)) {
                    stableMasks.set(i, fresh); matched = true; break;
                }
            }
            if (!matched && stableMasks.size() < MAX_MASKS) stableMasks.add(fresh);
        }
    }
    private void refreshContactsIfNeeded() {
        long now = SystemClock.uptimeMillis();
        if (now - lastContactRefreshAt < CONTACT_REFRESH_MS || repository == null) return;
        try { repository.invalidate(); repository.refreshNow(); lastContactRefreshAt = now; } catch (RuntimeException ignored) { }
    }
    private void beginTransitionCurtain() { beginTransitionCurtain(TRANSITION_CURTAIN_MS); }
    private void beginTransitionCurtain(long duration) {
        transitionCurtain = true;
        handler.removeCallbacks(clearCurtain);
        handler.postDelayed(clearCurtain, duration);
        render();
    }
    private void endTransitionCurtain() { if (!transitionCurtain) return; transitionCurtain = false; handler.removeCallbacks(clearCurtain); }
    private boolean isPhoneOrContactsPackage(String packageName) {
        if (packageName == null) return false;
        String v = packageName.toLowerCase(Locale.ROOT);
        return v.contains("contacts") || v.contains("dialer") || v.contains("incall") || v.contains("telecom") || v.contains("calllog") || v.contains("people") || v.equals("com.android.phone") || v.endsWith(".phone");
    }

    private void ensureOverlayAttached() {
        if (windowManager == null) return;
        if (maskLayerView == null) maskLayerView = new MaskLayerView();
        if (overlayAttached) return;
        WindowManager.LayoutParams lp = new WindowManager.LayoutParams(WindowManager.LayoutParams.MATCH_PARENT, WindowManager.LayoutParams.MATCH_PARENT, WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY,
                WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE | WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL | WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN | WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS | WindowManager.LayoutParams.FLAG_SECURE,
                PixelFormat.TRANSLUCENT);
        lp.gravity = Gravity.TOP | Gravity.START;
        if (Build.VERSION.SDK_INT >= 28) lp.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES;
        try { windowManager.addView(maskLayerView, lp); overlayAttached = true; } catch (RuntimeException ignored) { overlayAttached = false; }
    }
    private void render() { ensureOverlayAttached(); if (maskLayerView != null) maskLayerView.setState(stableMasks, transitionCurtain, isScrolling(), scrollAxis); }
    private void detachOverlay() {
        stableMasks.clear(); scrollGuardUntil = 0L; transitionCurtain = false;
        handler.removeCallbacks(activeMotionScan); handler.removeCallbacks(healthScan); handler.removeCallbacks(clearCurtain);
        if (overlayAttached && windowManager != null && maskLayerView != null) { try { windowManager.removeViewImmediate(maskLayerView); } catch (RuntimeException ignored) { } }
        overlayAttached = false;
    }

    @Override public void onConfigurationChanged(Configuration newConfig) {
        super.onConfigurationChanged(newConfig);
        stableMasks.clear();
        if (AppPreferences.isPrivacyShieldEnabled(this)) {
            if (isPhoneOrContactsPackage(foregroundPackage)) beginTransitionCurtain(350L);
            scheduleScan(0L); scheduleScan(120L);
        }
    }
    @Override public void onInterrupt() { if (AppPreferences.isPrivacyShieldEnabled(this)) scheduleScan(0L); else detachOverlay(); }
    @Override public void onDestroy() { handler.removeCallbacksAndMessages(null); detachOverlay(); super.onDestroy(); }

    private static final class MaskRegion { final Rect bounds; MaskRegion(Rect b) { bounds = b; } }
    private final class MaskLayerView extends View {
        private final Paint bg = new Paint(Paint.ANTI_ALIAS_FLAG);
        private final Paint text = new Paint(Paint.ANTI_ALIAS_FLAG);
        private List<MaskRegion> masks = new ArrayList<>();
        private boolean curtain;
        private boolean scrolling;
        private int axis;
        MaskLayerView() {
            super(PrivacyAccessibilityService.this);
            setBackgroundColor(Color.TRANSPARENT);
            setImportantForAccessibility(View.IMPORTANT_FOR_ACCESSIBILITY_NO);
            setFocusable(false); setClickable(false); setWillNotDraw(false);
            bg.setColor(Color.rgb(17,24,39)); bg.setStyle(Paint.Style.FILL);
            text.setColor(Color.WHITE); text.setTypeface(android.graphics.Typeface.DEFAULT_BOLD);
        }
        void setState(List<MaskRegion> source, boolean showCurtain, boolean moving, int scrollAxis) {
            masks = new ArrayList<>(source); curtain = showCurtain; scrolling = moving; axis = scrollAxis; invalidate();
        }
        @Override protected void onDraw(Canvas canvas) {
            super.onDraw(canvas);
            if (curtain) { canvas.drawColor(Color.rgb(17,24,39)); drawCentered(canvas, "Privacy shield checking…"); return; }
            for (MaskRegion region : masks) {
                Rect b = region.bounds;
                RectF area;
                if (scrolling && axis == AXIS_VERTICAL) area = new RectF(Math.max(0,b.left-12), 0, Math.min(getWidth(),b.right+12), getHeight());
                else if (scrolling && axis == AXIS_HORIZONTAL) area = new RectF(0, Math.max(0,b.top-10), getWidth(), Math.min(getHeight(),b.bottom+10));
                else area = new RectF(Math.max(0,b.left-7), Math.max(0,b.top-5), Math.min(getWidth(),b.right+7), Math.min(getHeight(),b.bottom+5));
                if (area.width() <= 1 || area.height() <= 1) continue;
                canvas.drawRoundRect(area, 6f, 6f, bg);
                if (!scrolling) drawMaskLabel(canvas, area);
            }
        }
        private void drawMaskLabel(Canvas canvas, RectF area) {
            String label = PhoneMasker.MASK;
            float size = Math.max(11f, Math.min(20f, area.height()*0.46f));
            text.setTextSize(size);
            while (text.measureText(label) > area.width()-14f && text.getTextSize() > 9f) text.setTextSize(text.getTextSize()-1f);
            Paint.FontMetrics fm = text.getFontMetrics();
            float y = area.centerY() - (fm.ascent + fm.descent)/2f;
            canvas.drawText(label, area.left+7f, y, text);
        }
        private void drawCentered(Canvas canvas, String message) {
            text.setTextSize(16f * getResources().getDisplayMetrics().scaledDensity);
            Paint.FontMetrics fm = text.getFontMetrics();
            float x = Math.max(24f, (getWidth()-text.measureText(message))/2f);
            float y = getHeight()/2f - (fm.ascent+fm.descent)/2f;
            canvas.drawText(message, x, y, text);
        }
    }
}
